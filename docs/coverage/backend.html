package calc

import (
        "math/big"
        "regexp"
        "strings"
)

const (
        maxOperandLen   = 32
        maxResultBits   = 4096
        formatPrecision = 12
)

var decimalPattern = regexp.MustCompile(`^-?\d+(\.\d+)?$`)

// ParseDecimal strictly parses a decimal string into an exact rational.
func ParseDecimal(s string) (*big.Rat, error) {
        if len(s) > maxOperandLen || !decimalPattern.MatchString(s) {
                return nil, newError(CodeInvalidOperand, "operand is not a valid decimal number")
        }
        // decimalPattern only admits strings that SetString always accepts.
        r, _ := new(big.Rat).SetString(s)
        return r, nil
}

// FormatDecimal renders r with up to 12 decimal places, trailing zeros trimmed.
func FormatDecimal(r *big.Rat) (string, error) {
        if r.Num().BitLen() > maxResultBits || r.Denom().BitLen() > maxResultBits {
                return "", newError(CodeOutOfRange, "result is out of range")
        }
        s := r.FloatString(formatPrecision)
        s = strings.TrimRight(s, "0")
        s = strings.TrimRight(s, ".")
        if s == "-0" {
                return "0", nil
        }
        return s, nil
}

// Code is a stable, machine-readable error identifier shared with the HTTP API.
type Code string

const (
        CodeInvalidJSON         Code = "INVALID_JSON"
        CodeUnknownOperation    Code = "UNKNOWN_OPERATION"
        CodeInvalidOperandCount Code = "INVALID_OPERAND_COUNT"
        CodeInvalidOperand      Code = "INVALID_OPERAND"
        CodeDivisionByZero      Code = "DIVISION_BY_ZERO"
        CodeNegativeSquareRoot  Code = "NEGATIVE_SQUARE_ROOT"
        CodeNonIntegerExponent  Code = "NON_INTEGER_EXPONENT"
        CodeOutOfRange          Code = "OUT_OF_RANGE"
        CodeMethodNotAllowed    Code = "METHOD_NOT_ALLOWED"
        CodePayloadTooLarge     Code = "PAYLOAD_TOO_LARGE"
)

// Error is a typed error carrying a stable Code.
type Error struct {
        Code    Code
        Message string
}

func (e *Error) Error() string { return e.Message }

func newError(code Code, msg string) *Error {
        return &Error{Code: code, Message: msg}
}

package calc

import "math/big"

// Add returns args[0] + args[1].
func Add(args []*big.Rat) (*big.Rat, error) {
        return new(big.Rat).Add(args[0], args[1]), nil
}

// Subtract returns args[0] - args[1].
func Subtract(args []*big.Rat) (*big.Rat, error) {
        return new(big.Rat).Sub(args[0], args[1]), nil
}

// Multiply returns args[0] * args[1].
func Multiply(args []*big.Rat) (*big.Rat, error) {
        return new(big.Rat).Mul(args[0], args[1]), nil
}

// Divide returns args[0] / args[1] exactly; rounding happens only on format.
func Divide(args []*big.Rat) (*big.Rat, error) {
        if args[1].Sign() == 0 {
                return nil, newError(CodeDivisionByZero, "cannot divide by zero")
        }
        return new(big.Rat).Quo(args[0], args[1]), nil
}

const maxExponent = 100

// Power returns args[0] ^ args[1]. The exponent must be an integer with
// absolute value at most 100; a negative exponent gives the reciprocal.
func Power(args []*big.Rat) (*big.Rat, error) {
        base, exp := args[0], args[1]
        if !exp.IsInt() {
                return nil, newError(CodeNonIntegerExponent, "exponent must be an integer")
        }
        if exp.Num().CmpAbs(big.NewInt(maxExponent)) > 0 {
                return nil, newError(CodeOutOfRange, "exponent must be between -100 and 100")
        }
        n := exp.Num().Int64()
        if n == 0 {
                return big.NewRat(1, 1), nil
        }
        if n < 0 {
                if base.Sign() == 0 {
                        return nil, newError(CodeDivisionByZero, "cannot raise zero to a negative power")
                }
                n = -n
                return new(big.Rat).SetFrac(pow(base.Denom(), n), pow(base.Num(), n)), nil
        }
        return new(big.Rat).SetFrac(pow(base.Num(), n), pow(base.Denom(), n)), nil
}

func pow(x *big.Int, n int64) *big.Int {
        return new(big.Int).Exp(x, big.NewInt(n), nil)
}

var (
        sqrtScale        = pow(big.NewInt(10), 14)
        sqrtScaleSquared = pow(big.NewInt(10), 28)
)

// Sqrt returns the square root of args[0] truncated to 14 decimal digits;
// FormatDecimal later rounds it to 12.
func Sqrt(args []*big.Rat) (*big.Rat, error) {
        x := args[0]
        if x.Sign() < 0 {
                return nil, newError(CodeNegativeSquareRoot, "cannot take the square root of a negative number")
        }
        scaled := new(big.Int).Mul(x.Num(), sqrtScaleSquared)
        scaled.Quo(scaled, x.Denom())
        root := scaled.Sqrt(scaled)
        return new(big.Rat).SetFrac(root, sqrtScale), nil
}

// Percent takes [p, base] and returns p percent of base: base * p / 100.
func Percent(args []*big.Rat) (*big.Rat, error) {
        r := new(big.Rat).Mul(args[1], args[0])
        return r.Quo(r, big.NewRat(100, 1)), nil
}

import "math/big"

// OperationFunc computes a result from already-parsed operands.
type OperationFunc func(args []*big.Rat) (*big.Rat, error)

// Operation describes one supported calculation.
type Operation struct {
        Name  string
        Arity int
        Fn    OperationFunc
}

// Registry maps an operation name to its definition.
type Registry map[string]Operation

func defaultRegistry() Registry {
        ops := []Operation{
                {Name: "add", Arity: 2, Fn: Add},
                {Name: "subtract", Arity: 2, Fn: Subtract},
                {Name: "multiply", Arity: 2, Fn: Multiply},
                {Name: "divide", Arity: 2, Fn: Divide},
                {Name: "power", Arity: 2, Fn: Power},
                {Name: "sqrt", Arity: 1, Fn: Sqrt},
                {Name: "percent", Arity: 2, Fn: Percent},
        }
        reg := make(Registry, len(ops))
        for _, op := range ops {
                reg[op.Name] = op
        }
        return reg
}

package calc

import (
        "fmt"
        "math/big"
        "sort"
)

// OperationInfo is the public description of a supported operation.
type OperationInfo struct {
        Name  string
        Arity int
}

// Service runs calculations on decimal strings.
type Service struct {
        reg Registry
}

func NewService() *Service {
        return &Service{reg: defaultRegistry()}
}

// Calculate looks up the operation, validates operands, runs it and formats the result.
func (s *Service) Calculate(op string, operands []string) (string, error) {
        operation, ok := s.reg[op]
        if !ok {
                return "", newError(CodeUnknownOperation, fmt.Sprintf("unknown operation %q", op))
        }
        if len(operands) != operation.Arity {
                return "", newError(CodeInvalidOperandCount,
                        fmt.Sprintf("operation %q expects %d operand(s), got %d", op, operation.Arity, len(operands)))
        }

        args := make([]*big.Rat, len(operands))
        for i, raw := range operands {
                r, err := ParseDecimal(raw)
                if err != nil {
                        return "", newError(CodeInvalidOperand,
                                fmt.Sprintf("operand %d is not a valid decimal number", i+1))
                }
                args[i] = r
        }

        result, err := operation.Fn(args)
        if err != nil {
                return "", err
        }
        return FormatDecimal(result)
}

// Operations lists supported operations sorted by name.
func (s *Service) Operations() []OperationInfo {
        infos := make([]OperationInfo, 0, len(s.reg))
        for _, op := range s.reg {
                infos = append(infos, OperationInfo{Name: op.Name, Arity: op.Arity})
        }
        sort.Slice(infos, func(i, j int) bool { return infos[i].Name < infos[j].Name })
        return infos
}

import (
        "encoding/json"
        "errors"
        "log/slog"
        "net/http"

        "github.com/Valentina17varela/sezzle-calculator/backend/internal/calc"
)

const codeInternalError calc.Code = "INTERNAL_ERROR"

const internalErrorMessage = "internal server error"

type errorBody struct {
        Code    calc.Code `json:"code"`
        Message string    `json:"message"`
}

type errorEnvelope struct {
        Error errorBody `json:"error"`
}

// statusFor maps an error code to its HTTP status. Unmapped codes are 500.
func statusFor(code calc.Code) int {
        switch code {
        case calc.CodeInvalidJSON,
                calc.CodeUnknownOperation,
                calc.CodeInvalidOperandCount,
                calc.CodeInvalidOperand:
                return http.StatusBadRequest
        case calc.CodeDivisionByZero,
                calc.CodeNegativeSquareRoot,
                calc.CodeNonIntegerExponent,
                calc.CodeOutOfRange:
                return http.StatusUnprocessableEntity
        case calc.CodeMethodNotAllowed:
                return http.StatusMethodNotAllowed
        case calc.CodePayloadTooLarge:
                return http.StatusRequestEntityTooLarge
        default:
                return http.StatusInternalServerError
        }
}

// writeError writes the standard JSON error envelope.
func writeError(w http.ResponseWriter, status int, code calc.Code, msg string) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(errorEnvelope{Error: errorBody{Code: code, Message: msg}})
}

// writeCalcError maps err to the error envelope. Anything that is not a
// mapped *calc.Error becomes a generic 500; the details go to the log only.
func writeCalcError(w http.ResponseWriter, err error) {
        var calcErr *calc.Error
        if errors.As(err, &calcErr) {
                if status := statusFor(calcErr.Code); status != http.StatusInternalServerError {
                        writeError(w, status, calcErr.Code, calcErr.Message)
                        return
                }
        }
        slog.Error("internal error", "error", err)
        writeError(w, http.StatusInternalServerError, codeInternalError, internalErrorMessage)
}

package httpapi

import (
        "encoding/json"
        "errors"
        "io"
        "net/http"

        "github.com/Valentina17varela/sezzle-calculator/backend/internal/calc"
)

type handler struct {
        svc *calc.Service
}

type calculateRequest struct {
        Operation string   `json:"operation"`
        Operands  []string `json:"operands"`
}

type calculateResponse struct {
        Operation string   `json:"operation"`
        Operands  []string `json:"operands"`
        Result    string   `json:"result"`
}

func (h *handler) calculate(w http.ResponseWriter, r *http.Request) {
        var req calculateRequest
        if err := decodeBody(r, &req); err != nil {
                writeCalcError(w, err)
                return
        }

        result, err := h.svc.Calculate(req.Operation, req.Operands)
        if err != nil {
                writeCalcError(w, err)
                return
        }

        writeJSON(w, http.StatusOK, calculateResponse{
                Operation: req.Operation,
                Operands:  req.Operands,
                Result:    result,
        })
}

// decodeBody reads exactly one JSON value into dst and rejects unknown fields.
func decodeBody(r *http.Request, dst any) error {
        dec := json.NewDecoder(r.Body)
        dec.DisallowUnknownFields()
        if err := dec.Decode(dst); err != nil {
                return bodyError(err)
        }

        var extra json.RawMessage
        err := dec.Decode(&extra)
        if err == io.EOF {
                return nil
        }
        if err == nil {
                err = errors.New("unexpected data after JSON value")
        }
        return bodyError(err)
}

func bodyError(err error) *calc.Error {
        var tooLarge *http.MaxBytesError
        if errors.As(err, &tooLarge) {
                return &calc.Error{Code: calc.CodePayloadTooLarge, Message: "request body is too large"}
        }
        return &calc.Error{
                Code:    calc.CodeInvalidJSON,
                Message: "request body must be a single JSON object with a string operation and string operands",
        }
}

type operationItem struct {
        Name  string `json:"name"`
        Arity int    `json:"arity"`
}

type operationsResponse struct {
        Operations []operationItem `json:"operations"`
}

type healthResponse struct {
        Status string `json:"status"`
}

func (h *handler) operations(w http.ResponseWriter, r *http.Request) {
        infos := h.svc.Operations()
        items := make([]operationItem, len(infos))
        for i, info := range infos {
                items[i] = operationItem{Name: info.Name, Arity: info.Arity}
        }
        writeJSON(w, http.StatusOK, operationsResponse{Operations: items})
}

func (h *handler) health(w http.ResponseWriter, r *http.Request) {
        writeJSON(w, http.StatusOK, healthResponse{Status: "ok"})
}

// methodNotAllowed answers any method not registered for a known path.
func methodNotAllowed(allow string) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                w.Header().Set("Allow", allow)
                writeError(w, http.StatusMethodNotAllowed, calc.CodeMethodNotAllowed, "method not allowed")
        }
}

func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(v)
}

import (
        "bytes"
        "errors"
        "io"
        "log/slog"
        "net/http"
        "runtime/debug"
        "time"

        "github.com/Valentina17varela/sezzle-calculator/backend/internal/calc"
)

const maxBodyBytes = 1024

// withMiddleware wraps next with request logging, panic recovery and the body
// size limit. Logging is outermost so a recovered panic is logged as a 500.
func withMiddleware(logger *slog.Logger, next http.Handler) http.Handler {
        return logRequests(logger, recoverPanics(logger, limitBody(maxBodyBytes, next)))
}

type statusRecorder struct {
        http.ResponseWriter
        status      int
        wroteHeader bool
}

func (s *statusRecorder) WriteHeader(code int) {
        if !s.wroteHeader {
                s.status = code
                s.wroteHeader = true
        }
        s.ResponseWriter.WriteHeader(code)
}

func (s *statusRecorder) Write(b []byte) (int, error) {
        s.wroteHeader = true
        return s.ResponseWriter.Write(b)
}

func logRequests(logger *slog.Logger, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                start := time.Now()
                rec := &statusRecorder{ResponseWriter: w, status: http.StatusOK}

                next.ServeHTTP(rec, r)

                logger.Info("request",
                        "method", r.Method,
                        "path", r.URL.Path,
                        "status", rec.status,
                        "duration", time.Since(start),
                )
        })
}

func recoverPanics(logger *slog.Logger, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                defer func() {
                        p := recover()
                        if p == nil {
                                return
                        }
                        if p == http.ErrAbortHandler {
                                panic(p)
                        }
                        logger.Error("panic recovered", "panic", p, "stack", string(debug.Stack()))
                        writeError(w, http.StatusInternalServerError, codeInternalError, internalErrorMessage)
                }()
                next.ServeHTTP(w, r)
        })
}

// limitBody reads the whole body up front, so an oversized request is always
// a 413 and never a parse error found earlier in the stream.
func limitBody(limit int64, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                body, err := io.ReadAll(http.MaxBytesReader(w, r.Body, limit))
                if err != nil {
                        var tooLarge *http.MaxBytesError
                        if errors.As(err, &tooLarge) {
                                writeError(w, http.StatusRequestEntityTooLarge, calc.CodePayloadTooLarge, "request body is too large")
                                return
                        }
                        writeError(w, http.StatusBadRequest, calc.CodeInvalidJSON, "could not read request body")
                        return
                }
                r.Body = io.NopCloser(bytes.NewReader(body))
                next.ServeHTTP(w, r)
        })
}

package httpapi

import (
        "log/slog"
        "net/http"

        "github.com/Valentina17varela/sezzle-calculator/backend/internal/calc"
)

// NewRouter wires the API routes and middleware to the given calculation service.
func NewRouter(svc *calc.Service) http.Handler {
        return withMiddleware(slog.Default(), newMux(svc))
}

func newMux(svc *calc.Service) *http.ServeMux {
        h := &handler{svc: svc}
        mux := http.NewServeMux()
        mux.HandleFunc("POST /api/v1/calculate", h.calculate)
        mux.HandleFunc("GET /api/v1/operations", h.operations)
        mux.HandleFunc("GET /health", h.health)

        // Method-less patterns are less specific, so they only catch other methods.
        mux.HandleFunc("/api/v1/calculate", methodNotAllowed("POST"))
        mux.HandleFunc("/api/v1/operations", methodNotAllowed("GET, HEAD"))
        mux.HandleFunc("/health", methodNotAllowed("GET, HEAD"))
        return mux
}
